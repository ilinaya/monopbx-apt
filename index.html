sudo install -d -m 0755 /etc/apt/keyrings
curl -fsSL https://ilinaya.github.io/monopbx-apt/key.gpg | sudo tee /etc/apt/keyrings/monopbx.gpg >/dev/null
echo "deb [arch=amd64 signed-by=/etc/apt/keyrings/monopbx.gpg] https://ilinaya.github.io/monopbx-apt stable main" | sudo tee /etc/apt/sources.list.d/monopbx.list
sudo apt-get update && sudo apt-get install -y monopbx
sudo monopbx setup --domain pbx.example.com --email admin@example.com
